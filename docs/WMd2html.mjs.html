//imgConvertToBase64為true時, 內嵌於輸出html之圖片base64字元總量(各data URL長度之合計, 同圖多處引用逐處計)上限: 2^27字元, 約96MiB之PNG
//  依據(2026-10-29, Node v24.22.1, 1600x900雜訊PNG每張0.52MiB): 圖片不經DOM後, 處理之峰值記憶體隨base64總字元數線性成長(RSS約為其4.5~5倍, 堆積約1~2倍),
//  245張(1.78億字元)於堆積上限576MB下通過, 300張(2.18億字元)於堆積上限576MB下OOM, 於704MB下通過; 本上限(1.34億字元)於堆積上限576MB下有約1.3倍餘裕,
//  與字串長度上限(V8, 536,870,888字元)相距甚遠; 超過者請改用imgConvertToBase64: false
//  修前(圖片隨html進DOMPurify與jsdom, 堆積約為字元數之35倍): 預設堆積4288MB下約80~100MB之圖片即OOM中止行程(審計md2docx-run01 E1b)
const IMG_BASE64_MAX_CHARS = 2 ** 27


/**
 * Markdown檔轉Html檔
 *
 * @param {String} fpIn 輸入來源Markdown檔位置字串
 * @param {String} fpOut 輸入轉出Html檔位置字串
 * @param {Object} [opt={}] 輸入設定物件，預設{}
 * @param {Boolean} [opt.linkBlank=true] 輸入連結是否為另開新頁布林值，預設true
 * @param {Boolean} [opt.tableHorizontalAlignmentCenter=true] 輸入表格是否水平置中布林值，預設true
 * @param {String} [opt.tableBorderColor='#666'] 輸入表格邊框顏色 (CSS color)
 * @param {string[]} [opt.fontFamilies=['Microsoft JhengHei','Avenir','Helvetica','Arial','sans-serif']] 輸入內文字型優先順序列表
 * @param {String} [opt.fontSizeUnit='pt'] 輸入字型大小單位字串，可使用例如'pt'、'px'等，預設'pt'
 * @param {Number} [opt.fontSizeScale=1] 輸入字型大小縮放倍率數字，預設1
 * @param {Number|String} [opt.fontSizeDef] 輸入內文字型大小數字或字串，預設為12，自動轉字型為12*fontSizeScale+fontSizeUnit
 * @param {Number|String} [opt.fontSizeH1] 輸入h1字型大小數字或字串，預設為20，自動轉字型為20*fontSizeScale+fontSizeUnit
 * @param {Number|String} [opt.fontSizeH2] 輸入h2字型大小數字或字串，預設為16，自動轉字型為16*fontSizeScale+fontSizeUnit
 * @param {Number|String} [opt.fontSizeH3] 輸入h3字型大小數字或字串，預設為14，自動轉字型為14*fontSizeScale+fontSizeUnit
 * @param {Number|String} [opt.fontSizeH4] 輸入h4字型大小數字或字串，預設為12，自動轉字型為12*fontSizeScale+fontSizeUnit
 * @param {Number|String} [opt.fontSizeH5] 輸入h5字型大小數字或字串，預設為12，自動轉字型為12*fontSizeScale+fontSizeUnit
 * @param {Number|String} [opt.fontSizeH6] 輸入h6字型大小數字或字串，預設為12，自動轉字型為12*fontSizeScale+fontSizeUnit
 * @param {Number|String} [opt.fontSizeP] 輸入p字型大小數字或字串，預設為12，自動轉字型為12*fontSizeScale+fontSizeUnit
 * @param {Number|String} [opt.fontSizeTab] 輸入表格文字字型大小數字或字串，預設為11，自動轉字型為11*fontSizeScale+fontSizeUnit
 * @param {Number|String} [opt.fontSizeCode] 輸入程式碼區塊字型大小數字或字串，預設為10，自動轉字型為10*fontSizeScale+fontSizeUnit
 * @param {String} [opt.textAlignH1='left'] 輸入h1對齊方式，可使用'left'、'center'、'right'，預設'left'
 * @param {String} [opt.textAlignH2='left'] 輸入h2對齊方式，可使用'left'、'center'、'right'，預設'left'
 * @param {String} [opt.textAlignH3='left'] 輸入h3對齊方式，可使用'left'、'center'、'right'，預設'left'
 * @param {String} [opt.textAlignH4='left'] 輸入h4對齊方式，可使用'left'、'center'、'right'，預設'left'
 * @param {String} [opt.textAlignH5='left'] 輸入h5對齊方式，可使用'left'、'center'、'right'，預設'left'
 * @param {String} [opt.textAlignH6='left'] 輸入h6對齊方式，可使用'left'、'center'、'right'，預設'left'
 * @param {Number|String} [opt.imgWidthMax=null] 輸入圖片樣式給予最大寬度，可輸入數字500單位為px，或是字串例如'100%'，預設null
 * @param {String} [opt.htmlTemp=null] 輸入模板字串，其內須取代{title}與{html}，預設null
 * @param {Boolean} [opt.imgConvertToBase64=true] 輸入本機圖片是否自動轉Base64內嵌布林值，預設true。md語法圖與原始<img>標籤之src皆以md所在資料夾解析路徑(絕對路徑則照用)，原始<img>之src值可為雙引號、單引號或無引號，同一行(同一段文字)可有任意多個，只處理<img>標籤之src屬性(data-src與其他標籤之src不處理)，程式碼區塊與行內程式碼內之文字不視為圖片引用；具通訊協定(如https:、data:)之網址不處理。true時輸出為自足html，各圖重新編碼為PNG內嵌，圖片總量過大(base64合計超過上限)則reject。false時不讀取、不轉碼，圖片src改寫為本機絕對路徑(Windows為反斜線形式，Word可解析；正斜線與百分比編碼不可)，輸出html不可攜，僅供同機之後續處理(如轉Word)。兩種模式下，圖片檔不存在皆reject，訊息列出全部缺圖路徑
 * @param {Function} [opt.funProcFpOut=null] 輸入處理輸出檔名函數，函數會傳入{fpOut,pretitle}，分別為原始輸出檔位置與提取Markdown內pretitle區文字，預設null
 * @returns {Promise} 回傳Promise，resolve回傳成功訊息，reject回傳錯誤訊息
 * @example
 *
 * import w from 'wsemi'
 * import WMd2html from './src/WMd2html.mjs'
 * //import WMd2html from 'w-md2html/src/WMd2html.mjs'
 * //import WMd2html from 'w-md2html'
 *
 * async function test() {
 *
 *     let fpIn = `./test/report.md`
 *     let fpOut = `./test/report.html`
 *     let opt = {
 *         imgWidthMax: '500px',
 *         funProcFpOut: (msg) => {
 *             console.log('msg', msg)
 *             return msg.fpOut
 *         },
 *     }
 *
 *     let r = await WMd2html(fpIn, fpOut, opt)
 *     console.log(r)
 *     // => ok
 *
 *     w.fsDeleteFile(fpOut)
 *
 * }
 * test()
 *     .catch((err) => {
 *         console.log('catch', err)
 *     })
 *
 */
async function WMd2html(fpIn, fpOut, opt = {}) {

    //linkBlank
    let linkBlank = get(opt, 'linkBlank', null)
    if (!isbol(linkBlank)) {
        linkBlank = true
    }

    //tableHorizontalAlignmentCenter
    let tableHorizontalAlignmentCenter = get(opt, 'tableHorizontalAlignmentCenter', null)
    if (!isbol(tableHorizontalAlignmentCenter)) {
        tableHorizontalAlignmentCenter = true
    }

    //check
    if (!fsIsFile(fpIn)) {
        return Promise.reject(`fpIn[${fpIn}] does not exist`)
    }

    //轉絕對路徑
    fpIn = path.resolve(fpIn)
    // console.log('fpIn', fpIn)

    //fdIn
    let fdIn = getPathParent(fpIn)

    //htmlTemp
    let htmlTemp = get(opt, 'htmlTemp', null)
    if (!isestr(htmlTemp)) {
        htmlTemp = `
<!DOCTYPE html>
<html>
<head>
    <meta charset="utf-8" />
    <meta http-equiv="X-UA-Compatible" content="IE=edge" />
    <meta http-equiv="Cache-Control" content="no-siteapp"/>
    <meta name="renderer" content="webkit" />
    <meta name="viewport" content="width=device-width, initial-scale=1, maximum-scale=1, user-scalable=no" />
    <title>{title}</title>
</head>
<body style="padding:20px; margin:0; overflow-y:auto;">
    {html}
</body>
</html>
`
    }

    //imgConvertToBase64
    let imgConvertToBase64 = get(opt, 'imgConvertToBase64', null)
    if (!isbol(imgConvertToBase64)) {
        imgConvertToBase64 = true
    }

    //funProcFpOut
    let funProcFpOut = get(opt, 'funProcFpOut')

    let getPretitle = (md) => {
        let s = split(md, '\n')
        let pretitle = ''
        each(s, (v, k) => {
            if (v.indexOf(' pretitle ') >= 0) {
                pretitle = get(s, k + 1, '') //取有pretitle的下一列
                pretitle = trim(pretitle) //因整行須trim
                return false //跳出
            }
        })
        return pretitle
    }

    //以註解或標籤為單位由左而右掃描html字串之掃描式(取圖表名之getItemName與登記圖片引用之cvPicRef共用), 供new RegExp(reTag, 'g')使用:
    //  註解整段為一個單位(未閉合者至字串結尾), 標籤名稱至空白,/,>為止(群組1), 其後屬性部分(群組2)可含引號包住之>與<, 故註解內或其他標籤屬性值內之<img字樣不會被當成標籤
    let reTag = String.raw`<!--[\s\S]*?(?:-->|$)|<([A-Za-z][^ \t\n\f\r/>]*)((?:[^>"']|"[^"]*"|'[^']*')*)>`

    //找屬性, attrs為標籤名稱之後的屬性字串, 回傳第一個名稱(不分大小寫, 須完整相等)為name者{value: 屬性值, index: 該屬性(含名稱, 等號與值)於attrs之起點, length: 其長度}, 無此屬性回傳null
    //  屬性值可為雙引號, 單引號或無引號(至空白或>為止, 含/與=), 可含>與另一種引號; 屬性不帶值時value為空字串; 空白只認html之空白字元(不含U+00A0), 與html解析器一致
    let findAttr = (attrs, name) => {
        let reAttr = /([^ \t\n\f\r"'<>/=]+)(?:[ \t\n\f\r]*=[ \t\n\f\r]*(?:"([^"]*)"|'([^']*)'|([^ \t\n\f\r>]+)))?/g
        let m = reAttr.exec(attrs)
        while (m !== null) {
            if (m[1].toLowerCase() === name) {
                return { value: m[2] ?? m[3] ?? m[4] ?? '', index: m.index, length: m[0].length }
            }
            m = reAttr.exec(attrs)
        }
        return null
    }

    //取屬性值, 無此屬性回傳null, 屬性不帶值回傳空字串
    let getAttr = (attrs, name) => {
        let a = findAttr(attrs, name)
        return a === null ? null : a.value
    }

    //取圖表名, 只取「圖n」「表n」所在之文字: 圖片與圖題寫在同一行時, 有src之<img>須先移除, 否則html2str會將其轉為「alt [src]」併入圖表名, 使圖表名對不上內文之{圖表名}
    //  以註解或標籤為單位由左而右掃描(屬性值可含<與>), 故註解內或其他標籤屬性值內的「<img」字樣不會被當成圖片; 移除圖片後留下之無文字連結(<a href><img></a>)亦移除, 否則html2str會輸出其href
    //  無src(含src為空)之<img>維持原狀(html2str僅輸出alt), 沒有移除任何圖片時該行與修正前完全相同; 有移除圖片時, 若轉為純文字後已不含「圖n」(原本只在被移除之img屬性或其他標籤之屬性內)則非圖題, 回傳null不登記, 避免以空名稱或他字覆寫真正之圖題
    let getItemName = async (v, type) => {
        let rm = false
        let t = v.replace(new RegExp(reTag, 'g'), (m, name, attrs) => {
            if (name !== undefined && name.toLowerCase() === 'img' && getAttr(attrs, 'src')) {
                rm = true
                return '\u0001' //以記號標示被移除之圖片位置, 供下方判斷連結是否因此變空
            }
            return m
        })
        if (rm) {
            //連結之內容只剩空白, 標籤, 註解與被移除之圖片(至少一個)時, 連結一併移除; 原本就沒有文字之連結(與圖片無關者)不動
            let sk = String.raw`[ \t\n\f\r]|<!--[\s\S]*?-->|<\/?[A-Za-z](?:[^>"']|"[^"]*"|'[^']*')*>`
            t = t.replace(new RegExp(String.raw`<a(?:[ \t\n\f\r](?:[^>"']|"[^"]*"|'[^']*')*)?>(?:${sk})*\u0001(?:${sk}|\u0001)*<\/a[ \t\n\f\r]*>`, 'gi'), '')
            t = t.replaceAll('\u0001', '')
        }
        t = await html2str(t)
        if (rm && t.indexOf(`${type}n`) < 0) {
            return null
        }
        t = replace(t, `${type}n`, '')
        t = trim(t)
        return `{${t}}`
    }

    //取得輸出(最終html)之可見文字中仍殘留之{…}(即圖表編號與模板符號取代後仍對不上任何圖表名者): 以渲染後之html為準, 移除程式碼(code, pre)、數學式(math, katex)、style與script後, 於區塊邊界以換行分隔之文字取出
    //  文字節點內之空白(含軟換行)依html規則視為單一空白; 被移除之元素與區塊邊界以換行取代, 故不會拼出原文不存在之字串, 也不會將跨越程式碼或區塊之大括號視為一對
    //  屬性值(alt, title, href)不是可見文字不檢查; 逸出之大括號在輸出中是可見文字故會列出; math與katex: KaTeX同時輸出MathML與顯示層(後者含字面之大括號, 如\{x\}); style: 套件自身之樣式含選項值(如fontFamilies), 可能出現單行之{…}; script: 使用者內容中者已於md2html清洗時移除, 自訂模板(htmlTemp)中者仍在
    //  inline以外之元素皆視為區塊(清洗後只剩標準html/svg元素, 預設顯示皆已知), 不依CSS實際顯示
    let getUnresolvedRefs = async (html) => {
        let inlineTags = new Set(['a', 'abbr', 'acronym', 'audio', 'b', 'bdi', 'bdo', 'big', 'button', 'canvas', 'cite', 'data', 'del', 'dfn', 'em', 'embed', 'font', 'i', 'iframe', 'img', 'input', 'ins', 'kbd', 'label', 'mark', 'meter', 'nobr', 'object', 'output', 'picture', 'progress', 'q', 'rp', 'rt', 'ruby', 's', 'samp', 'select', 'small', 'source', 'span', 'strike', 'strong', 'sub', 'sup', 'svg', 'textarea', 'time', 'tt', 'u', 'var', 'video', 'wbr'])
        let clib = 'jsdom'
        let { JSDOM } = await import(clib)
        let { window } = new JSDOM(html)
        let doc = window.document
        let box = doc.body
        let tw = doc.createTreeWalker(box, window.NodeFilter.SHOW_TEXT)
        let nds = []
        while (tw.nextNode()) {
            nds.push(tw.currentNode)
        }
        each(nds, (nd) => {
            nd.nodeValue = nd.nodeValue.replace(/[ \t\n\f\r]+/g, ' ')
        })
        each(box.querySelectorAll('code, pre, style, script, math, .katex, .katex-error'), (el) => {
            el.replaceWith('\n')
        })
        each(box.querySelectorAll('*'), (el) => {
            if (!inlineTags.has(el.localName)) {
                el.before('\n')
                el.after('\n')
            }
        })
        let txt = box.textContent.replace(/ {2,}/g, ' ') //相鄰文字節點各自折疊後可能留下連續空白
        window.close()
        return uniq(txt.match(/\{[^{}\n]*[^{}\s][^{}\n]*\}/g) || [])
    }

    let cvItemNum = async(md, type) => {
        let s = split(md, '\n') //不能用sep避免md內手動分行消失
        let i = 0
        let rs = []
        let kp = {}
        await pmSeries(s, async (v) => {
            let vv = v
            if (v.indexOf(`${type}n`) >= 0) {
                i++

                //圖表編號
                let nn = `${type}${i}`

                //更新成圖表編號
                vv = replace(v, `${type}n`, nn)

                //儲存圖表名對應之圖表編號
                let t = await getItemName(v, type)
                if (t !== null) {
                    kp[t] = nn
                }
                // console.log(t, '-->', kp[t])

            }

            //儲存圖表編號
            rs.push(vv)

        })

        //取代圖表名為對應圖表編號
        let h = join(rs, '\n')
        each(kp, (v, k) => {
            // h = replace(h, k, v)
            h = h.replaceAll(k, v)
        })

        return h
    }

    //本機圖片引用: md語法圖(image token)與原始<img>標籤(html token, cvPicRef)皆於walkTokens只登記並暫以佔位符取代src, 待md2html完成後再依imgConvertToBase64一次寫入(fillImgs)
    //  why 佔位符: ①DOMPurify會移除值為Windows絕對路徑(C:\...)之src(視為未知通訊協定), 絕對路徑只能於清洗之後寫入; ②圖片之data URL不經marked, DOMPurify與jsdom, 其堆積不隨圖片總量放大(清洗後堆積約為html字元數之35倍)
    //  路徑一律以md所在資料夾解析(絕對路徑照用), 同一檔共用一個佔位符; 空值, 具通訊協定(https:, data:, file:等), 協定相對(//)與錨點(#)者非本機圖, 維持原值不處理
    let phImg = '' //佔位符前綴, 讀入md後決定
    let fpsImg = [] //已登記之圖片絕對路徑, 佔位符之編號即其索引
    let kpFpImg = {} //絕對路徑對應之編號
    let regImg = (ref) => {
        if (!isestr(ref) || /^(?:[A-Za-z][A-Za-z0-9+.-]+:|\/\/|#)/.test(ref)) { //協定至少2字元, 故磁碟代號(C:\)不誤判為協定
            return null
        }
        let fp = path.resolve(fdIn, ref)
        if (kpFpImg[fp] === undefined) {
            kpFpImg[fp] = size(fpsImg)
            fpsImg.push(fp)
        }
        return `${phImg}${kpFpImg[fp]}`
    }

    //原始<img>標籤之src: 於walkTokens對html token(區塊與行內)之text呼叫, 故程式碼區塊與行內程式碼(非html token)內之文字不處理
    //  以getItemName同一掃描式(reTag)逐標籤處理, 一段文字可有任意多個<img>(不分行, 標籤屬性可跨行); 只處理<img>標籤(不分大小寫)之src屬性, 屬性名稱須完整相等(data-src, 他標籤之src皆不是, 註解內之<img>亦不處理)
    //  src值可為雙引號, 單引號或無引號(依html解析器規則, 見findAttr), 同一標籤有多個src時只取第一個(與html解析器一致); 本機圖整個src屬性改寫為src="佔位符"(一律雙引號), 之後與md語法圖同一流程由fillImgs填入; 非本機圖, 空值與無src者原樣保留
    let cvPicRef = (html) => {
        return html.replace(new RegExp(reTag, 'g'), (m, name, attrs) => {
            if (name === undefined || name.toLowerCase() !== 'img') {
                return m
            }
            let a = findAttr(attrs, 'src')
            if (a === null) {
                return m
            }
            let ph = regImg(a.value)
            if (ph === null) {
                return m
            }
            return `<${name}${attrs.slice(0, a.index)}src="${ph}"${attrs.slice(a.index + a.length)}>`
        })
    }

    let fillImgs = async (h) => {

            }

            each(ids, (i) => {

            let nChar = 0

                nChar += size(urls[i]) * cnt[i]
                if (nChar > IMG_BASE64_MAX_CHARS) {
                    throw new Error(msgOver(nChar, nb, false))
                }
